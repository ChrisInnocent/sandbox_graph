def audit_collection(actual, expected):
    """Both inputs must be nonempty, unique and exactly equal as sets."""
    if not actual or not expected:
        raise ValueError('empty collection or reference')
    if len(actual) != len(set(actual)) or len(expected) != len(set(expected)):
        raise ValueError('duplicate nodeid')
    if set(actual) != set(expected):
        raise ValueError('missing or unexpected nodeid')

    ('missing-parameter', ['case[a]'], ['case[a]', 'case[b]']),
    ('unexpected', ['case[a]', 'case[b]'], ['case[a]']),
]:
    try:
        audit_collection(actual, expected)
    except ValueError:
        negative_checks.append({'case': label, 'rejected': True})
    else:
        raise AssertionError('invalid collection was accepted: ' + label)
